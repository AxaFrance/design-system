## Link

### Import

```tsx
import { Link } from "@axa-fr/canopee-react/prospect";
```

### Use

```tsx
export const LinkComponent = () => (
  <Link openInNewTab href="https://fakelink.com">
    Plus de détails
  </Link>
);
```

### React Router

The `as` prop selects the rendered component and infers its props and ref type.
By default, `Link` renders an HTML anchor. React Router must be installed in the
application, and this example must be rendered inside a router.

```tsx
import { Link as RouterLink } from "react-router";
import { Link } from "@axa-fr/canopee-react/prospect";

export const AccountLink = () => (
  <Link as={RouterLink} to="/account">
    Mon compte
  </Link>
);
```
